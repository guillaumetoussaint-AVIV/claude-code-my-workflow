\documentclass{beamer}
\usetheme{Madrid}
\usepackage{graphicx}
\graphicspath{{../Figures/}}

% -----------------------------------------------------------------------------
% Palette Université Paris Dauphine–PSL
%
% dauphine-blue is the exact RGB fill operator read out of the official
% Dauphine–PSL doctoral-thesis cover template (page 1 content stream), NOT from
% a published charte graphique. The two tints below are Beamer mixes derived
% from it, not institutional colours.
% -----------------------------------------------------------------------------
\definecolor{dauphine-blue}{HTML}{24388D}   % institutional blue
\definecolor{dauphine-light}{HTML}{E9EBF4}  % derived tint, block backgrounds

\setbeamercolor{structure}{fg=dauphine-blue}
\setbeamercolor{palette primary}{bg=dauphine-blue,fg=white}
\setbeamercolor{palette secondary}{bg=dauphine-blue!80,fg=white}
\setbeamercolor{palette tertiary}{bg=dauphine-blue!60,fg=white}
\setbeamercolor{titlelike}{parent=palette primary}
\setbeamercolor{frametitle}{bg=dauphine-blue,fg=white}
\setbeamercolor{block title}{bg=dauphine-blue,fg=white}
\setbeamercolor{block body}{bg=dauphine-light,fg=black}
